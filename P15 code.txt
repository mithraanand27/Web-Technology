      const switches = document.querySelectorAll('.switch input[type="checkbox"]');

      function paint(sw) {                                               // update the label, row style and page class
        const key = sw.dataset.setting;
        $('state-' + key).textContent = sw.checked ? 'On' : 'Off';
        sw.closest('.setting').classList.toggle('is-on', sw.checked);
        document.documentElement.classList.toggle('setting-' + key, sw.checked);
      }
      function announce(detail) { document.dispatchEvent(new CustomEvent('voltsettings', {detail})); }

      switches.forEach(sw => {
        const key = sw.dataset.setting;
        sw.checked = !!state[key]; paint(sw);                            // restore the saved position
        sw.addEventListener('change', () => {
          state[key] = sw.checked; paint(sw); save();
          announce({key, value: sw.checked});
          if (window.showToast) window.showToast(NAMES[key] + (sw.checked ? ' turned on' : ' turned off'));
        });
      });

      $('resetSettings').addEventListener('click', () => {
        switches.forEach(sw => { if (sw.checked) { sw.checked = false; sw.dispatchEvent(new Event('change')); } });
        save();
      });

      announce({init: true});                                            // apply saved settings to the rest of the page
    })();

          t.setAttribute('tabindex', on ? '0' : '-1');
