    (function () {
      const toast = document.getElementById('toast'), toastMsg = document.getElementById('toastMsg');
      let toastTimer;

      // Notification: slide in, then hide itself after 3.5 seconds
      function showToast(text) {
        clearTimeout(toastTimer);
        toast.classList.remove('show');
        void toast.offsetWidth;                       // restart the animation
        toastMsg.textContent = text;
        toast.classList.add('show');
        toastTimer = setTimeout(() => toast.classList.remove('show'), 3500);
      }
      toast.addEventListener('click', () => { clearTimeout(toastTimer); toast.classList.remove('show'); });

      // Vehicle cards: open / close the quick-specs panel
      document.querySelectorAll('.quick-btn').forEach(btn => {
        btn.addEventListener('click', () => {
          const panel = document.getElementById(btn.getAttribute('aria-controls'));
          const open = panel.classList.toggle('open');
          btn.setAttribute('aria-expanded', open);
          btn.textContent = open ? 'Hide specs' : 'Quick specs';
        });
      });

      // Vehicle cards: save to shortlist (heart pops, toast confirms)
      document.querySelectorAll('.save-btn').forEach(btn => {
        btn.addEventListener('click', () => {
          const saved = btn.classList.toggle('saved');
          btn.setAttribute('aria-pressed', saved);
          btn.querySelector('.heart').innerHTML = saved ? '&#9829;' : '&#9825;';
          showToast(btn.dataset.name + (saved ? ' added to your shortlist' : ' removed from your shortlist'));
        });
      });

      // FAQ accordion: one answer open at a time
      const faqItems = document.querySelectorAll('.faq-item');
      faqItems.forEach(item => {
        const q = item.querySelector('.faq-q');
        q.addEventListener('click', () => {
          const willOpen = !item.classList.contains('open');
          faqItems.forEach(o => { o.classList.remove('open'); o.querySelector('.faq-q').setAttribute('aria-expanded', 'false'); });
          if (willOpen) { item.classList.add('open'); q.setAttribute('aria-expanded', 'true'); }
        });
      });

      // Charging simulation: 10% to 80% in timed steps
      const demo = document.getElementById('chargeDemo'), pct = document.getElementById('pct'),
            cmsg = document.getElementById('chargeMsg'), cbtn = document.getElementById('chargeBtn');
      let timers = [];
      function resetCharge() {
        timers.forEach(clearTimeout); timers = [];
        demo.classList.remove('charging', 'done');
        pct.textContent = '10%';
        cmsg.textContent = 'Press start to simulate a 10-80% fast charge.';
        cbtn.textContent = 'Start fast charge';
      }
      cbtn.addEventListener('click', () => {
        if (demo.classList.contains('charging') || demo.classList.contains('done')) { resetCharge(); return; }
        demo.classList.add('charging');
        cbtn.textContent = 'Stop and reset';
        cmsg.textContent = 'Charging at up to 250 kW...';
        for (let i = 1; i <= 10; i++) {
          timers.push(setTimeout(() => {
            pct.textContent = (10 + i * 7) + '%';
            if (i === 5) cmsg.textContent = 'Nearly halfway there...';
            if (i === 10) {
              demo.classList.remove('charging'); demo.classList.add('done');
              cmsg.textContent = 'Charged to 80%. A 10-80% charge takes 18 to 22 minutes.';
              cbtn.textContent = 'Run again';
              showToast('Fast charge complete: 80%');
            }
          }, i * 600));
        }
      });

      // Booking form: notifications
      const form = document.getElementById('enquiryForm');
      form.addEventListener('submit', () => showToast('Test drive request received. We will confirm your slot soon.'));
      form.addEventListener('reset', () => showToast('Form cleared'));
    })();
  </script>
